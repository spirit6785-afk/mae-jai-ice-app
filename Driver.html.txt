  <script>
    // ==========================================
    // 🧠 GLOBAL DRIVER STATE
    // ==========================================
    let DRV = {
      pin: '',
      driver: {},
      lockedRoute: null,
      myDispatch: null,
      selectedCustomerId: '',
      walkInCustomName: '',
      walkInPresetMode: 'NORMAL',
      walkInPriceDelta: 0,
      walkInPromo3For100: false,
      walkInCustomPrices: {},
      walkInGroupKeys: [], // 🚶 เก็บรายการคีย์ของลูกค้าขาจรแยกตามชื่อแต่ละงาน
      shopFilterMode: 'PENDING',
      shopSearchQuery: '',
      cart: {},
      payType: 'เงินสด',
      truckInventory: [],
      customerSalesMap: {},
      reloadHistory: [],
      finance: {
        cashSales: 0, transferSales: 0, creditSales: 0, totalSales: 0,
        collectedDebtCash: 0, collectedDebtTransfer: 0, cashFuelExpense: 0,
        netCashToSubmit: 0, creditBills: []
      },
      data: {
        otherActiveRoutes: [], vehicles: [], products: [], customers: [],
        specialPrices: [], promoMap: {}, customerDebtMap: {},
        todayDebtPayments: [], todayRouteSales: [], todayTransfers: [], todayFuelLogs: []
      }
    };

    function esc(str) {
      return String(str || '').replace(/\\/g, '\\\\').replace(/'/g, "\\'").replace(/"/g, '&quot;');
    }

    function closeDrvModal() {
      document.getElementById('drv-modal-backdrop').classList.add('hidden');
    }

    function toast(msg, type = 'success') {
      const box = document.getElementById('toast-box');
      const el = document.createElement('div');
      el.className = `${type === 'success' ? 'bg-emerald-600' : 'bg-rose-600'} text-white p-3.5 rounded-2xl shadow-xl font-bold text-sm text-center transition-all`;
      el.innerText = msg;
      box.appendChild(el);
      setTimeout(() => el.remove(), 3000);
    }

    function callApi(action, payload = {}) {
      document.getElementById('loading').classList.remove('hidden');
      return new Promise((resolve, reject) => {
        if (typeof google !== 'undefined' && google.script && google.script.run) {
          google.script.run.withSuccessHandler(res => {
            document.getElementById('loading').classList.add('hidden');
            if (res && res.status === 'success') resolve(res.data);
            else { toast(res ? res.message : 'เกิดข้อผิดพลาด', 'error'); reject(res); }
          }).withFailureHandler(err => {
            document.getElementById('loading').classList.add('hidden');
            toast(err.message, 'error');
            reject(err);
          }).processApiRequest({ action, payload });
        } else {
          document.getElementById('loading').classList.add('hidden');
          toast('กรุณาเปิดใช้งานผ่านลิงก์ Web App', 'error');
          reject('Not in GAS');
        }
      });
    }

    // ==========================================
    // 🔐 LOGIN & SYNC DATA
    // ==========================================
    async function verifyDriverPin() {
      const pin = document.getElementById('login-pin').value.trim();
      if (!pin) return toast('กรุณากรอกรหัส PIN 4 หลัก', 'error');
      try {
        const d = await callApi('getDriverInitData', { pin });
        DRV.pin = pin;
        DRV.driver = d.driver || {};
        DRV.lockedRoute = d.lockedRoute;
        DRV.myDispatch = d.myDispatch;
        DRV.data = d;
        DRV.shopFilterMode = 'PENDING';

        document.getElementById('login-screen').classList.add('hidden');
        document.getElementById('app-screen').classList.remove('hidden');

        computeRouteAndTruckState();
        switchTab('sale');

        if (DRV.myDispatch && (!parseFloat(DRV.myDispatch.StartMileage) || parseFloat(DRV.myDispatch.StartMileage) <= 0)) {
          openMileageModal('START');
        }
      } catch (e) {}
    }

    function logoutDriver() {
      DRV.pin = '';
      DRV.cart = {};
      DRV.selectedCustomerId = '';
      DRV.walkInCustomName = '';
      DRV.walkInCustomPrices = {};
      document.getElementById('login-pin').value = '';
      document.getElementById('app-screen').classList.add('hidden');
      document.getElementById('login-screen').classList.remove('hidden');
    }

    async function reloadDriverData() {
      try {
        const d = await callApi('getDriverInitData', { pin: DRV.pin });
        DRV.driver = d.driver || {};
        DRV.lockedRoute = d.lockedRoute;
        DRV.myDispatch = d.myDispatch;
        DRV.data = d;
        computeRouteAndTruckState();
        switchTab(DRV.currentTab || 'sale');
        toast(DRV.myDispatch ? 'ซิงค์ข้อมูลล่าสุดเรียบร้อย' : 'เสมียนปิดกะสายส่งนี้เรียบร้อยแล้วค่ะ');
      } catch (e) {}
    }

    // ==========================================
    // 🧮 COMPUTE TRUCK INVENTORY, LOAD HISTORY & SEPARATE WALK-IN GROUPS
    // ==========================================
    function computeRouteAndTruckState() {
      const dsp = DRV.myDispatch;
      const routeSales = DRV.data.todayRouteSales || [];
      const transfers = DRV.data.todayTransfers || [];
      const fuelLogs = DRV.data.todayFuelLogs || [];
      const debtMap = DRV.data.customerDebtMap || {};
      const allProds = DRV.data.products || [];

      document.getElementById('drv-name-label').innerText = `คนขับ: ${DRV.driver.EmpName || '-'}`;

      const debtShopCount = Object.keys(debtMap).length;
      const navDebtBadge = document.getElementById('nav-debt-badge');
      if (navDebtBadge) {
        navDebtBadge.innerText = debtShopCount;
        navDebtBadge.classList.toggle('hidden', debtShopCount === 0);
      }

      if (!dsp || !DRV.lockedRoute) {
        DRV.truckInventory = [];
        DRV.customerSalesMap = {};
        DRV.walkInGroupKeys = [];
        DRV.reloadHistory = [];
        DRV.cart = {};
        DRV.selectedCustomerId = '';
        DRV.walkInCustomName = '';
        DRV.walkInCustomPrices = {};
        DRV.finance = {
          cashSales: 0, transferSales: 0, creditSales: 0, totalSales: 0,
          collectedDebtCash: 0, collectedDebtTransfer: 0, cashFuelExpense: 0,
          netCashToSubmit: 0, creditBills: []
        };

        document.getElementById('locked-route-badge').innerText = '🏁 ปิดกะแล้ว / รอปล่อยรถ';
        document.getElementById('drv-vehicle-badge').innerText = '';
        document.getElementById('stat-remain-qty').innerText = '0 หน่วย';
        document.getElementById('stat-shop-progress').innerText = '0 / 0 ร้าน';
        document.getElementById('stat-cash-net').innerText = '฿0';
        document.getElementById('mileage-status-bar').innerHTML = `<p class="text-xs text-emerald-400 font-extrabold w-full text-center py-1">✅ เสมียนปิดกะสายส่งนี้เรียบร้อยแล้ว (รอปล่อยรถกะถัดไป)</p>`;
        document.getElementById('truck-stock-pills').innerHTML = '';
        return;
      }

      document.getElementById('locked-route-badge').innerText = `🔒 ${DRV.lockedRoute.RouteName}`;
      document.getElementById('drv-vehicle-badge').innerText = `(${dsp.VehicleID || '-'})`;

      const noteParts = String(dsp.Note || '').split('|').map(s => s.trim()).filter(s => s !== '');
      DRV.reloadHistory = noteParts.map((part, idx) => {
        const isInitial = idx === 0 && part.startsWith('เบิก:');
        const isReload = part.includes('เบิกเพิ่ม');
        const isTransfer = part.includes('โอนให้') || part.includes('รับโอนจาก');
        return {
          step: idx + 1,
          type: isInitial ? 'INITIAL' : (isReload ? 'RELOAD' : (isTransfer ? 'TRANSFER' : 'NOTE')),
          label: isInitial ? `🚀 เบิกกะเช้า (${String(dsp.DispatchDate || '').slice(11, 16) || 'เปิดกะ'})` : (isReload ? '➕ เบิกเพิ่มระหว่างวัน' : (isTransfer ? '🔄 โอนย้ายระหว่างสาย' : '📝 หมายเหตุ')),
          detail: part.replace(/^เบิก:\s*/, '')
        };
      });

      const startMile = parseFloat(dsp.StartMileage) || 0;
      const endMile = parseFloat(dsp.EndMileage) || 0;
      const dist = (startMile > 0 && endMile >= startMile) ? (endMile - startMile) : 0;

      document.getElementById('mileage-status-bar').innerHTML = `
        <div onclick="openMileageModal('START')" class="flex-1 bg-slate-900/90 px-3 py-1.5 rounded-xl border ${startMile > 0 ? 'border-slate-700' : 'border-amber-400 animate-pulse'} cursor-pointer active:scale-95 transition">
          <p class="text-[9px] text-slate-400">1. เลขไมล์ก่อนออกส่ง</p>
          <p class="text-xs font-extrabold ${startMile > 0 ? 'text-white' : 'text-amber-400'}">${startMile > 0 ? startMile.toLocaleString() + ' กม.' : '⚠️ กดลงไมล์เริ่ม'}</p>
        </div>
        <i data-lucide="arrow-right" class="w-4 h-4 text-slate-500 shrink-0"></i>
        <div onclick="openMileageModal('END')" class="flex-1 bg-slate-900/90 px-3 py-1.5 rounded-xl border ${endMile > 0 ? 'border-emerald-500/60' : 'border-slate-700'} cursor-pointer active:scale-95 transition">
          <p class="text-[9px] text-slate-400">2. ไมล์จบ ${dist > 0 ? `(วิ่ง ${dist} กม.)` : ''}</p>
          <p class="text-xs font-extrabold ${endMile > 0 ? 'text-emerald-400' : 'text-blue-400'}">${endMile > 0 ? endMile.toLocaleString() + ' กม.' : '🏁 กดลงไมล์จบ'}</p>
        </div>
      `;

      const soldByProd = {};
      const custSalesMap = {};
      const walkInKeys = [];
      let cashSales = 0, transferSales = 0, creditSales = 0;
      const creditBills = [];

      // 🧾 รวบรวมบิลขายแยกตามร้านค้า (สำคัญ: หากเป็นขาจร ให้แยก Group Key ตามชื่อที่คนขับพิมพ์ เพื่อไม่ให้งานศพ 2 งานถูกยุบรวมกัน!)
      routeSales.forEach(s => {
        if (s.PaymentStatus === 'Cancelled' || s.PaymentStatus === 'ยกเลิก') return;
        const amt = parseFloat(s.TotalAmount) || 0;
        const pType = String(s.PaymentType || '');
        if (pType.includes('เครดิต') || pType.includes('บิลเชื่อ') || pType.includes('ค้างชำระ')) {
          creditSales += amt;
          creditBills.push(s);
        } else if (pType.includes('โอน')) {
          transferSales += amt;
        } else {
          cashSales += amt;
        }

        const rawCid = String(s.CustomerID || 'WALK-IN').trim();
        const rawCName = String(s.CustomerName || 'ลูกค้าขาจรทั่วไป').trim();
        const isWalkInSale = (rawCid === 'WALK-IN' || rawCid === '');
        const groupKey = isWalkInSale ? `WALK-IN::${rawCName}` : rawCid;

        if (!custSalesMap[groupKey]) {
          custSalesMap[groupKey] = {
            CustomerID: groupKey,
            RawCustomerID: rawCid,
            CustomerName: rawCName,
            IsWalkIn: isWalkInSale,
            count: 0,
            totalAmt: 0,
            cashAmt: 0,
            trfAmt: 0,
            crdAmt: 0,
            totalQty: 0,
            lastTime: String(s.SaleDate || '').slice(11, 16),
            summaries: [],
            bills: []
          };
          if (isWalkInSale) walkInKeys.push(groupKey);
        }
        custSalesMap[groupKey].count++;
        custSalesMap[groupKey].totalAmt += amt;
        custSalesMap[groupKey].totalQty += (parseFloat(s.TotalQty) || 0);
        if (pType.includes('เครดิต') || pType.includes('บิลเชื่อ') || pType.includes('ค้างชำระ')) custSalesMap[groupKey].crdAmt += amt;
        else if (pType.includes('โอน')) custSalesMap[groupKey].trfAmt += amt;
        else custSalesMap[groupKey].cashAmt += amt;

        if (s.ItemSummary) custSalesMap[groupKey].summaries.push(`${s.ItemSummary} [${s.PaymentType}]`);

        custSalesMap[groupKey].bills.push({
          SaleID: s.SaleID,
          SaleTime: String(s.SaleDate || '').slice(11, 16),
          SaleDate: String(s.SaleDate || '').slice(0, 16),
          CustomerName: rawCName,
          ItemSummary: s.ItemSummary || '-',
          PaymentType: s.PaymentType || 'เงินสด',
          TotalQty: parseFloat(s.TotalQty) || 0,
          TotalAmount: amt
        });

        (s.Items || []).forEach(item => {
          soldByProd[item.ProductID] = (soldByProd[item.ProductID] || 0) + (parseFloat(item.Quantity) || 0);
        });
      });
      DRV.customerSalesMap = custSalesMap;
      DRV.walkInGroupKeys = walkInKeys;

      const transferByProd = {};
      const transferNameByProd = {};
      transfers.forEach(t => {
        const tQty = parseFloat(t.Quantity) || 0;
        const pid = String(t.ProductID);
        if (t.ProductName) transferNameByProd[pid] = t.ProductName;
        if (String(t.ToDispatchID) === String(dsp.DispatchID)) transferByProd[pid] = (transferByProd[pid] || 0) + tQty;
        else if (String(t.FromDispatchID) === String(dsp.DispatchID)) transferByProd[pid] = (transferByProd[pid] || 0) - tQty;
      });

      let loadedItemsMap = {};
      try {
        const parsed = JSON.parse(dsp.LoadDetails || '[]');
        if (Array.isArray(parsed)) parsed.forEach(ld => loadedItemsMap[String(ld.id)] = { name: ld.name, qty: parseFloat(ld.qty) || 0 });
      } catch (e) {}

      const allProductIds = new Set([...Object.keys(loadedItemsMap), ...Object.keys(transferByProd), ...Object.keys(soldByProd)]);
      const truckInv = [];
      let totalRemainQty = 0;

      allProductIds.forEach(pid => {
        if (!pid || pid === 'FEE' || pid === 'DISCOUNT' || pid === 'AR-MANUAL') return;
        const masterProd = allProds.find(p => String(p.ProductID) === String(pid)) || {
          ProductID: pid,
          ProductName: (loadedItemsMap[pid]?.name || transferNameByProd[pid] || pid),
          DefaultPrice: 0,
          Unit: 'กระสอบ'
        };
        const lQty = loadedItemsMap[pid] ? loadedItemsMap[pid].qty : 0;
        const trfNet = transferByProd[pid] || 0;
        const sQty = soldByProd[pid] || 0;
        const effectiveLoaded = (loadedItemsMap[pid] !== undefined && lQty > 0) ? lQty : Math.max(0, trfNet);
        const rQty = effectiveLoaded - sQty;
        if (effectiveLoaded <= 0 && rQty <= 0 && sQty <= 0) return;

        totalRemainQty += Math.max(0, rQty);
        truckInv.push({
          ProductID: masterProd.ProductID,
          ProductName: masterProd.ProductName,
          DefaultPrice: parseFloat(masterProd.DefaultPrice) || 0,
          Unit: masterProd.Unit || 'กระสอบ',
          loadedQty: effectiveLoaded,
          soldQty: sQty,
          remainQty: rQty
        });
      });
      DRV.truckInventory = truckInv;

      let cashFuelExpense = 0;
      fuelLogs.forEach(f => {
        if (String(f.PaidBy || '').includes('เงินสด') || String(f.PaidBy || '').includes('หัก')) {
          cashFuelExpense += (parseFloat(f.TotalCost) || 0);
        }
      });

      let collectedDebtCash = 0, collectedDebtTransfer = 0;
      (DRV.data.todayDebtPayments || []).forEach(p => {
        const pAmt = parseFloat(p.AmountPaid) || 0;
        if (String(p.PaymentMethod || '').includes('โอน')) collectedDebtTransfer += pAmt;
        else collectedDebtCash += pAmt;
      });

      const netCashToSubmit = cashSales + collectedDebtCash - cashFuelExpense;
      DRV.finance = {
        cashSales, transferSales, creditSales,
        totalSales: cashSales + transferSales + creditSales,
        collectedDebtCash, collectedDebtTransfer,
        cashFuelExpense, netCashToSubmit, creditBills
      };

      const myCustomers = DRV.data.customers || [];
      const deliveredCount = myCustomers.filter(c => custSalesMap[c.CustomerID]).length;
      const walkInCount = walkInKeys.length;

      document.getElementById('stat-remain-qty').innerText = `${totalRemainQty} หน่วย`;
      document.getElementById('stat-shop-progress').innerText = `${deliveredCount}/${myCustomers.length} ร้าน${walkInCount > 0 ? ` (+${walkInCount} ขาจร)` : ''}`;
      document.getElementById('stat-cash-net').innerText = `฿${netCashToSubmit.toLocaleString()}`;

      document.getElementById('truck-stock-pills').innerHTML = truckInv.length === 0
        ? `<div class="col-span-2 bg-slate-800 text-slate-400 p-2 rounded-xl text-xs text-center">ไม่มีรายการน้ำแข็งบนรถ</div>`
        : truckInv.map(item => `
          <div class="bg-slate-800/95 px-3 py-2 rounded-2xl border ${item.remainQty <= 0 ? 'border-rose-500/50 opacity-75' : 'border-slate-700'} flex justify-between items-center">

              <p class="text-xs font-extrabold text-white truncate">${item.ProductName}</p>

            </div>
            <div class="text-right shrink-0">
              <span class="text-base font-black ${item.remainQty <= 0 ? 'text-rose-400' : 'text-amber-400'} block leading-none">${item.remainQty}</span>
              <span class="text-[9px] text-slate-400">${item.Unit}</span>
            </div>
          </div>
        `).join('');
    }

    function isWalkInSelection_(cid) {

    function getCurrentWalkInLookupKey_() {
      if (!isWalkInSelection_(DRV.selectedCustomerId)) return '';
      if (String(DRV.selectedCustomerId).startsWith('WALK-IN::')) return DRV.selectedCustomerId;
      const custom = (DRV.walkInCustomName || '').trim();
      const fullName = custom ? (custom.startsWith('ขาจร:') ? custom : `ขาจร: ${custom}`) : 'ลูกค้าขาจรทั่วไป';
      return `WALK-IN::${fullName}`;

function getDriverPrice(prod) {
      const pid = String(prod.ProductID);
      const defaultPrice = parseFloat(prod.DefaultPrice) || 0;

      // 🚶 กรณีลูกค้าขาจร (WALK-IN)
      if (isWalkInSelection_(DRV.selectedCustomerId)) {
        if (!DRV.walkInCustomPrices) DRV.walkInCustomPrices = {};
        // 1. หากคนขับพิมพ์หรือกดปรับราคาเฉพาะรายการนี้ด้วยตัวเอง ให้ใช้ราคานั้นก่อน
        if (DRV.walkInCustomPrices[pid] !== undefined && DRV.walkInCustomPrices[pid] !== '') {
          return parseFloat(DRV.walkInCustomPrices[pid]) || 0;
        }
        // 2. หากเลือกโหมด "🎁 โปรคละ 3 ชิ้น 100.-" ให้ราคาเริ่มต้นที่ 35 บาททันที (1 ถุง=35, 2 ถุง=70, 3 ถุง=105 ลด 5 เหลือ 100)
        if (DRV.walkInPromo3For100) {
          return Math.max(35, defaultPrice);
        }
        // 3. หากเลือกโหมดบวก/ลบตามเรต
        if (DRV.walkInPriceDelta) {
          return Math.max(5, defaultPrice + (parseFloat(DRV.walkInPriceDelta) || 0));
        }
        return defaultPrice;
      }

      // 🏪 กรณีร้านค้าประจำในสายส่ง
      if (DRV.selectedCustomerId) {
        const sp = (DRV.data.specialPrices || []).find(s => String(s.CustomerID) === String(DRV.selectedCustomerId) && String(s.ProductID) === pid);
        if (sp) return parseFloat(sp.SpecialPrice);

        // หากร้านประจำเปิดสิทธิ์โปรคละ 3 ชิ้น 100.- (และไม่ได้ล็อกราคาพิเศษอื่นไว้) ให้ราคาต่อหน่วยเริ่มที่ 35 บาทเช่นกัน
        const custPromo = (DRV.data.promoMap || {})[DRV.selectedCustomerId];
        if (custPromo) {
          const eligStr = String(custPromo.EligibleProducts || 'ALL');
          const isElig = (eligStr === 'ALL' || eligStr.split(',').map(x => x.trim()).includes(pid));
          if (isElig) return Math.max(35, defaultPrice);
        }
      }
      return defaultPrice;
    }

    function setWalkInPresetMode(mode) {
      DRV.walkInPresetMode = mode;
      DRV.walkInCustomPrices = {}; // ล้างราคาที่เคยปรับมือ เพื่อให้ใช้ราคาตามโหมดที่เลือกทันที
      if (mode === 'NORMAL') {
        DRV.walkInPriceDelta = 0;
        DRV.walkInPromo3For100 = false;
      } else if (mode === 'PROMO_3_100') {
        DRV.walkInPriceDelta = 0;
        DRV.walkInPromo3For100 = true; // บังคับราคาเริ่มที่ 35 บาทใน getDriverPrice อัตโนมัติ
      } else if (mode === 'PLUS_5') {
        DRV.walkInPriceDelta = 5;
        DRV.walkInPromo3For100 = false;
      } else if (mode === 'MINUS_5') {
        DRV.walkInPriceDelta = -5;
        DRV.walkInPromo3For100 = false;
      }
      renderSaleTab();
      lucide.createIcons();
    }

    function adjWalkInItemPrice(pid, delta) {
      const prod = DRV.truckInventory.find(x => String(x.ProductID) === String(pid));
      if (!prod) return;
      if (!DRV.walkInCustomPrices) DRV.walkInCustomPrices = {};

    function calcDriverCartPricing() {
      const promoMap = DRV.data.promoMap || {};
      const isWalkIn = isWalkInSelection_(DRV.selectedCustomerId);
      const custPromo = (!isWalkIn && DRV.selectedCustomerId)
        ? promoMap[DRV.selectedCustomerId]
        : (isWalkIn && DRV.walkInPromo3For100 ? { EligibleProducts: 'ALL' } : null);

      const eligStr = custPromo ? String(custPromo.EligibleProducts || 'ALL') : '';
      const eligSet = (eligStr && eligStr !== 'ALL') ? new Set(eligStr.split(',').map(x => x.trim())) : null;

      let rawSubTotal = 0;
      let eligibleUnitPrices = [];

      DRV.truckInventory.forEach(p => {
        const qty = DRV.cart[p.ProductID] || 0;
        if (qty <= 0) return;
        const price = getDriverPrice(p);
        rawSubTotal += qty * price;
        if (custPromo && (!eligSet || eligSet.has(String(p.ProductID)))) {
          for (let i = 0; i < qty; i++) eligibleUnitPrices.push(price);
        }
      });

      const totalEligibleQty = eligibleUnitPrices.length;
      let bundleCount = 0;
      let discountAmt = 0;

      if (custPromo && totalEligibleQty >= 3) {
        eligibleUnitPrices.sort((a, b) => b - a);
        bundleCount = Math.floor(totalEligibleQty / 3);
        let normalPriceOfBundledItems = 0;
        for (let j = 0; j < bundleCount * 3; j++) {
          normalPriceOfBundledItems += eligibleUnitPrices[j];
        }
        if (normalPriceOfBundledItems > bundleCount * 100) {
          discountAmt = normalPriceOfBundledItems - (bundleCount * 100);
        }
      }

      const remMod = totalEligibleQty % 3;
      const needMore = remMod === 0 ? (totalEligibleQty > 0 ? 0 : 3) : (3 - remMod);

      return {
        hasPromo: !!custPromo,
        totalEligibleQty,
        bundleCount,
        needMore,
        rawSubTotal,
        discountAmt,
        netTotal: rawSubTotal - discountAmt
      };
    }

    function onSelectDrvCustomer(cid) {
      DRV.selectedCustomerId = cid;
      if (String(cid).startsWith('WALK-IN::')) {
        const rawName = String(cid).replace(/^WALK-IN::/, '').replace(/^ขาจร:\s*/, '');
        DRV.walkInCustomName = rawName === 'ลูกค้าขาจรทั่วไป' ? '' : rawName;
      } else if (cid === 'WALK-IN') {
        DRV.walkInCustomName = '';
      } else {
        DRV.walkInCustomName = '';
        DRV.walkInPresetMode = 'NORMAL';
        DRV.walkInPriceDelta = 0;
        DRV.walkInPromo3For100 = false;
        DRV.walkInCustomPrices = {};
      }
      renderSaleTab();
      lucide.createIcons();
      if (cid === 'WALK-IN') {
        setTimeout(() => document.getElementById('drv-walkin-name')?.focus(), 50);
      }
    }

    function selectWalkInQuick() {
      DRV.selectedCustomerId = 'WALK-IN';
      DRV.walkInCustomName = '';
      if (!DRV.walkInPresetMode) DRV.walkInPresetMode = 'NORMAL';
      renderSaleTab();
      lucide.createIcons();
      setTimeout(() => document.getElementById('drv-walkin-name')?.focus(), 50);
    }

    function selectExistingWalkInGroup(groupKey) {
      const info = DRV.customerSalesMap[groupKey] || (DRV.data.customerDebtMap || {})[groupKey];
      DRV.selectedCustomerId = groupKey;
      if (info) {
        const clean = String(info.CustomerName || '').replace(/^ขาจร:\s*/, '');
        DRV.walkInCustomName = clean === 'ลูกค้าขาจรทั่วไป' ? '' : clean;
      }
      switchTab('sale');
    }

    function adjDrvCart(pid, delta) {
      const item = DRV.truckInventory.find(x => x.ProductID === pid);
      if (!item) return;
      const next = Math.max(0, (DRV.cart[pid] || 0) + delta);
      if (delta > 0 && next > item.remainQty) return toast(`บนรถเหลือเพียง ${item.remainQty} ${item.Unit}`, 'error');
      if (next === 0) delete DRV.cart[pid];
      else DRV.cart[pid] = next;
      renderSaleTab();
      lucide.createIcons();
    }

    function setDrvCart(pid, val) {
      const item = DRV.truckInventory.find(x => x.ProductID === pid);
      if (!item) return;
      let q = Math.max(0, parseInt(val) || 0);
      if (q > item.remainQty) {
        toast(`บนรถเหลือเพียง ${item.remainQty} ${item.Unit}`, 'error');
        q = item.remainQty;
      }
      if (q === 0) delete DRV.cart[pid];
      else DRV.cart[pid] = q;
      renderSaleTab();
      lucide.createIcons();
    }

    function setDrvPayType(type) {
      DRV.payType = type;
      renderSaleTab();
      lucide.createIcons();
    }

    async function submitDrvSale() {
      if (!DRV.selectedCustomerId) return toast('กรุณาเลือกร้านค้า หรือกดปุ่ม "+ ขาจรรายใหม่" ก่อนบันทึกบิลค่ะ', 'error');

      const isWalkIn = isWalkInSelection_(DRV.selectedCustomerId);
      const customWalkIn = (document.getElementById('drv-walkin-name')?.value || DRV.walkInCustomName || '').trim();

      if (isWalkIn && DRV.payType.includes('เครดิต') && !customWalkIn) {
        document.getElementById('drv-walkin-name')?.focus();
        return toast('กรณีขาจร "ค้างชำระ" กรุณาพิมพ์ชื่อร้าน/ชื่องานและเบอร์โทรในช่องขาจรก่อนค่ะ!', 'error');
      }

      const items = [];
      Object.keys(DRV.cart).forEach(pid => {
        const p = DRV.truckInventory.find(x => x.ProductID === pid);
        if (p && DRV.cart[pid] > 0) {
          const soldPrice = getDriverPrice(p);
          const defaultPrice = parseFloat(p.DefaultPrice) || 0;
          const auditProdName = (isWalkIn && soldPrice !== defaultPrice)
            ? `${p.ProductName} (@฿${soldPrice})`
            : p.ProductName;

          items.push({
            ProductID: p.ProductID,
            ProductName: auditProdName,
            UnitPrice: soldPrice,
            Quantity: DRV.cart[pid],
            Unit: p.Unit
          });
        }
      });
      if (!items.length) return toast('กรุณากดจำนวนน้ำแข็งอย่างน้อย 1 รายการ', 'error');

      const pricing = calcDriverCartPricing();
      if (pricing.discountAmt > 0) {
        items.push({ ProductID: 'DISCOUNT', ProductName: `โปร 3 ชิ้น 100 (${pricing.bundleCount} ชุด)`, UnitPrice: -pricing.discountAmt, Quantity: 1, Unit: 'รายการ' });
      }

      const cust = (!isWalkIn) ? (DRV.data.customers || []).find(c => c.CustomerID === DRV.selectedCustomerId) : null;
      let finalCustName = 'ลูกค้าขาจรทั่วไป';
      if (isWalkIn) {
        if (customWalkIn) {
          finalCustName = customWalkIn.startsWith('ขาจร:') ? customWalkIn : `ขาจร: ${customWalkIn}`;
        } else {
          finalCustName = 'ลูกค้าขาจรทั่วไป';
        }
      } else if (cust) {
        finalCustName = cust.CustomerName;
      }

      try {
        await callApi('saveSale', {
          CustomerID: cust ? cust.CustomerID : 'WALK-IN',
          CustomerName: finalCustName,
          RouteID: DRV.lockedRoute.RouteID,
          PaymentType: DRV.payType,
          CreatedBy: 'คนขับ: ' + DRV.driver.EmpName,
          items: items
        });
        toast(`บันทึกบิล "${finalCustName}" สำเร็จ!`);
        DRV.cart = {};
        DRV.selectedCustomerId = '';
        DRV.walkInCustomName = '';
        DRV.walkInPresetMode = 'NORMAL';
        DRV.walkInPriceDelta = 0;
        DRV.walkInPromo3For100 = false;
        DRV.walkInCustomPrices = {};
        reloadDriverData();
      } catch (e) {}
    }

    async function submitDriverReload() {
      let addedQty = 0;
      const detailsArr = [];
      const detailsText = [];
      document.querySelectorAll('.drv-rld-qty').forEach(inp => {
        const q = parseFloat(inp.value) || 0;
        if (q > 0) {
          addedQty += q;
          detailsArr.push({ id: inp.getAttribute('data-id'), name: inp.getAttribute('data-name'), qty: q });
          detailsText.push(`${inp.getAttribute('data-name')} +${q}`);
        }
      });
      if (addedQty <= 0) return toast('กรุณาระบุจำนวนน้ำแข็งที่เบิกเพิ่ม', 'error');
      const note = document.getElementById('drv-rld-note')?.value.trim() || '';
      try {
        await callApi('reloadDispatch', {
          DispatchID: DRV.myDispatch.DispatchID,
          AddedQty: addedQty,
          AddedDetailsText: detailsText.join(', '),
          AddedDetailsJSON: JSON.stringify(detailsArr),
          Note: note,
          CreatedBy: DRV.driver.EmpName
        });
        toast('บันทึกเบิกน้ำแข็งเพิ่มสำเร็จ!');
        reloadDriverData();
      } catch (e) {}
    }

    async function submitIceTransfer() {
      const toSel = document.getElementById('trf-to-dsp');
      const prodSel = document.getElementById('trf-prod');
      const qty = parseFloat(document.getElementById('trf-qty').value) || 0;
      if (!toSel || !prodSel || qty <= 0) return toast('กรุณาเลือกข้อมูลและระบุจำนวนที่โอน', 'error');
      try {
        await callApi('saveIceTransfer', {
          FromDispatchID: DRV.myDispatch.DispatchID,
          ToDispatchID: toSel.value,
          ProductID: prodSel.value,
          ProductName: prodSel.options[prodSel.selectedIndex].getAttribute('data-name'),
          Quantity: qty,
          DriverName: DRV.driver.EmpName
        });
        toast('โอนน้ำแข็งให้เพื่อนร่วมสายสำเร็จ!');
        reloadDriverData();
      } catch (e) {}
    }

    function recalcDrvPay() {
      let sum = 0;
      document.querySelectorAll('.drv-pay-chk:checked').forEach(chk => {
        sum += parseFloat(chk.getAttribute('data-amt')) || 0;
      });
      const el = document.getElementById('drv-pay-amt-label');
      if (el) {
        el.innerText = '฿' + sum.toLocaleString();
        el.setAttribute('data-amt', sum);
      }
    }

    async function submitDrvDebtCollect(customerId, method) {
      const saleIds = [];
      document.querySelectorAll('.drv-pay-chk:checked').forEach(chk => saleIds.push(chk.value));
      if (saleIds.length === 0) return toast('กรุณาติ๊กเลือกบิลอย่างน้อย 1 บิล', 'error');

      const amount = parseFloat(document.getElementById('drv-pay-amt-label')?.getAttribute('data-amt')) || 0;
      const debtInfo = (DRV.data.customerDebtMap || {})[customerId];
      const custObj = (DRV.data.customers || []).find(c => String(c.CustomerID) === String(customerId));
      const custName = (debtInfo && debtInfo.CustomerName) || (custObj ? custObj.CustomerName : String(customerId).replace(/^WALK-IN::/, ''));
      const cleanCustomerId = String(customerId).startsWith('WALK-IN::') ? 'WALK-IN' : customerId;

      try {
        await callApi('savePayment', {
          CustomerID: cleanCustomerId,
          CustomerName: custName,
          SaleIDs: saleIds,
          AmountPaid: amount,
          PaymentMethod: method,
          CreatedBy: 'คนขับ: ' + DRV.driver.EmpName
        });
        closeDrvModal();
        toast('บันทึกรับชำระหนี้สำเร็จ!');
        reloadDriverData();
      } catch (e) {}
    }
