  function apiCall(action, payload = {}) {
    showLoading(true);

      }
    }
    return new Promise((resolve, reject) => {
      if (typeof google !== 'undefined' && google.script && google.script.run) {
        google.script.run.withSuccessHandler(res => {
          showLoading(false);
          if (res && res.status === 'success') resolve(res.data);
          else { showToast(res ? res.message : 'เกิดข้อผิดพลาดจากเซิร์ฟเวอร์', 'error'); reject(res ? res.message : 'Error'); }
        }).withFailureHandler(err => {
          showLoading(false); showToast('Error: ' + err.message, 'error'); reject(err);
        }).processApiRequest({ action, payload });
      } else { showLoading(false); showToast('กรุณาใช้งานผ่านลิงก์ Web App', 'error'); reject('Not in GAS'); }
    });
  }

  function parseIceBreakdownFromSales(salesList) {
    const counts = { bigTube: 0, crushed: 0, smallTube: 0, block: 0, retail: 0, other: 0 };
    let totalStockUnits = 0;
    (salesList || []).forEach(s => {
      if (s.PaymentStatus === 'Cancelled' || s.PaymentStatus === 'ยกเลิก') return;
      const summary = String(s.ItemSummary || ''); if (!summary || summary === '-') return;
      summary.split(',').forEach(part => {
        const p = part.trim(); if (!p || p.includes('ส่วนลด') || p.includes('โปร') || p.includes('ค่ากระสอบ') || p.includes('ชาร์จเพิ่ม') || p.includes('ยอดหนี้คงค้าง')) return;
        const match = p.match(/x\s*(-?\d+(\.\d+)?)$/i); const qty = match ? parseFloat(match[1]) : 1; if (qty <= 0) return;
        if (p.includes('ขายปลีก')) { counts.retail += qty; } else if (p.includes('หลอดใหญ่')) { counts.bigTube += qty; totalStockUnits += qty; } else if (p.includes('บด')) { counts.crushed += qty; totalStockUnits += qty; } else if (p.includes('หลอดเล็ก')) { counts.smallTube += qty; totalStockUnits += qty; } else if (p.includes('ก้อน') || p.includes('ซอง')) { counts.block += qty; totalStockUnits += qty; } else { counts.other += qty; totalStockUnits += qty; }
      });
    });
    const badges = [`หลอดใหญ่: <b class="text-white">${counts.bigTube.toLocaleString()}</b>`, `บด: <b class="text-white">${counts.crushed.toLocaleString()}</b>`, `หลอดเล็ก: <b class="text-white">${counts.smallTube.toLocaleString()}</b>`, `ก้อน: <b class="text-white">${counts.block.toLocaleString()}</b>`];
    if (counts.retail > 0) badges.push(`ขายปลีก: <b class="text-purple-300">${counts.retail.toLocaleString()}</b>`); if (counts.other > 0) badges.push(`อื่นๆ: <b class="text-white">${counts.other.toLocaleString()}</b>`);
    return { counts, totalStockUnits, inlineHTML: badges.join(' <span class="text-slate-500 mx-1">|</span> '), plainText: `หลอดใหญ่ ${counts.bigTube} | บด ${counts.crushed} | หลอดเล็ก ${counts.smallTube} | ก้อน ${counts.block}${counts.retail > 0 ? ' | ขายปลีก ' + counts.retail : ''}` };
  }

  function calcGrandTotal() {
    const items = Object.values(STATE.pos.cart);
    let subTotal = 0;
    let eligibleUnitPrices = [];

    const promoMap = STATE.pos.promoMap || {};
    const custPromo = STATE.pos.selectedCustomer ? promoMap[STATE.pos.selectedCustomer] : null;
    const eligStr = custPromo ? String(custPromo.EligibleProducts || 'ALL') : '';
    const eligSet = (eligStr && eligStr !== 'ALL') ? new Set(eligStr.split(',')) : null;

    items.forEach(i => {
      subTotal += i.Quantity * i.UnitPrice;
      if (custPromo && (!eligSet || eligSet.has(String(i.ProductID)))) {
        for (let j = 0; j < i.Quantity; j++) eligibleUnitPrices.push(i.UnitPrice);
      }
    });

    let autoDiscount = 0;
    if (custPromo && eligibleUnitPrices.length >= 3) {
      eligibleUnitPrices.sort((a, b) => b - a);
      const bundleCount = Math.floor(eligibleUnitPrices.length / 3);
      let normalPriceOfBundledItems = 0;
      for (let j = 0; j < bundleCount * 3; j++) normalPriceOfBundledItems += eligibleUnitPrices[j];
      if (normalPriceOfBundledItems > bundleCount * 100) autoDiscount = normalPriceOfBundledItems - (bundleCount * 100);
    }
    STATE.pos.autoPromoDiscount = autoDiscount;

    const promoBadge = document.getElementById('pos-promo-badge');
    if (promoBadge) {
      if (autoDiscount > 0) {
        promoBadge.innerHTML = `🎁 ส่วนลดโปร 3 ชิ้น 100: <span class="text-rose-600 font-black">-฿${autoDiscount.toLocaleString()}</span>`;
        promoBadge.classList.remove('hidden');
      } else {
        promoBadge.classList.add('hidden');
      }
    }
    
    const fee = parseFloat(document.getElementById('pos-fee')?.value) || 0;
    const discount = parseFloat(document.getElementById('pos-discount')?.value) || 0;
    const grandTotal = subTotal + fee - discount - autoDiscount;

    const totEl = document.getElementById('pos-tot');
    if (totEl) { totEl.innerText = '฿' + grandTotal.toLocaleString(); totEl.setAttribute('data-total', grandTotal); }
    
    const cashIn = parseFloat(document.getElementById('pos-cash-in')?.value) || 0;
    const change = cashIn - grandTotal;
    const changeEl = document.getElementById('pos-change');
    if (!changeEl) return;
    
    if (cashIn === 0) { changeEl.innerText = '฿0'; changeEl.className = 'font-bold text-slate-400'; } 
    else if (change < 0) { changeEl.innerText = 'ขาด ฿' + Math.abs(change).toLocaleString(); changeEl.className = 'font-bold text-rose-600'; } 
    else { changeEl.innerText = '฿' + change.toLocaleString(); changeEl.className = 'font-bold text-emerald-600 text-lg'; }
  }

  function adjCart(id, d) { 
    const retailItem = RETAIL_QUICK_ITEMS.find(x => x.ProductID === id); const p = retailItem || STATE.pos.products.find(x => x.ProductID === id); if (!p) return; 
    if (!retailItem) { const currentStock = STATE.pos.stockMap[id] || 0; const currentCartQty = STATE.pos.cart[id] ? STATE.pos.cart[id].Quantity : 0; if (d > 0 && (currentCartQty + d) > currentStock) showToast(`ระวัง! สต๊อกเหลือเพียง ${currentStock} ${p.Unit}`, 'error'); }
    const unitPrice = retailItem ? parseFloat(retailItem.DefaultPrice) : getPrice(p);
    if (!STATE.pos.cart[id]) STATE.pos.cart[id] = { ProductID: id, ProductName: p.ProductName, UnitPrice: unitPrice, Quantity: 0, Unit: p.Unit || 'รายการ' }; 
    STATE.pos.cart[id].Quantity += d; if (STATE.pos.cart[id].Quantity <= 0) delete STATE.pos.cart[id]; 
    renderPosProducts(); renderPosCart(); 
  }

  async function submitSale() {

    const tot = parseFloat(document.getElementById('pos-tot').getAttribute('data-total')) || 0;
    const payType = document.getElementById('pos-pay').value;
    const cashIn = parseFloat(document.getElementById('pos-cash-in').value) || 0;
    if (payType === 'เงินสด' && cashIn > 0 && cashIn < tot) return showToast('รับเงินมาไม่ครบตามจำนวนบิล', 'error');

    const fee = parseFloat(document.getElementById('pos-fee').value) || 0;
    const discount = parseFloat(document.getElementById('pos-discount').value) || 0;
    const autoDiscount = STATE.pos.autoPromoDiscount || 0;

    const finalItems = [...cartItems];
    if (fee > 0) finalItems.push({ ProductID: 'FEE', ProductName: 'ชาร์จเพิ่ม / ค่ากระสอบ', UnitPrice: fee, Quantity: 1, Unit: 'รายการ' });
    if (discount > 0) finalItems.push({ ProductID: 'DISCOUNT', ProductName: 'ส่วนลด', UnitPrice: -discount, Quantity: 1, Unit: 'รายการ' });
    if (autoDiscount > 0) finalItems.push({ ProductID: 'DISCOUNT', ProductName: 'ส่วนลดโปร 3 ชิ้น 100', UnitPrice: -autoDiscount, Quantity: 1, Unit: 'รายการ' });

    const custInputVal = document.getElementById('pos-cust-input').value;
    const cid = custInputVal ? custInputVal.split(' : ')[0].trim() : '';
    const rawBadgeName = document.getElementById('pos-cust-name-badge').innerText.replace('ลูกค้า: ', '').replace(' (ซื้อหน้าโรงงาน)', '');
    const payload = {
      CustomerID: cid || 'WALK-IN',
      CustomerName: cid ? rawBadgeName : 'ขาจร (หน้าโรงงาน)',
      RouteID: document.getElementById('pos-route').value,
      PaymentType: payType,
      CashTendered: payType === 'เงินสด' && cashIn > 0 ? cashIn : tot,
      Change: payType === 'เงินสด' && cashIn > 0 ? (cashIn - tot) : 0,
      items: finalItems
    };

    try { 
      const res = await apiCall('saveSale', payload);
      showToast('รับชำระเงินและบันทึกบิลสำเร็จ!');
      clearPosCart();
      if (document.getElementById('pos-cust-input')) document.getElementById('pos-cust-input').value = '';
      onPosCustDatalist('');
      await printSlip(res);
      await renderPOS();
    } catch(e) {} 
  }

  function calcDaysDiff(dateStr) {
    if (!dateStr) return 0; const d = new Date(String(dateStr).replace(' ', 'T')); if (isNaN(d.getTime())) return 0;
    return Math.max(0, Math.floor((new Date() - d) / (1000 * 60 * 60 * 24)));
  }

      const tDebt = parseFloat(c.TotalDebt) || 0;
      const cLimit = parseFloat(c.CreditLimit) || 0;

      calcTotalBills += (c.Bills ? c.Bills.length : 0);
      if (cLimit > 0 && tDebt > cLimit) calcOverLimit++;

  function drawCustomerTable() { 
    const view = document.getElementById('app-view'); 
    const searchQuery = document.getElementById('c-search') ? document.getElementById('c-search').value.toLowerCase() : ''; 
    const filterRoute = document.getElementById('c-filter-route') ? document.getElementById('c-filter-route').value : 'all'; 
    const inactiveMode = window.CUST_INACTIVE_FILTER || 'ALL';

    const allCusts = STATE.customerDashboard.list || [];

    // คำนวณจำนวนร้านในแต่ละกลุ่มความเสี่ยง (ตามสายส่งที่เลือก)
    const routeScopedCusts = allCusts.filter(c => filterRoute === 'all' || c.RouteID === filterRoute);
    const countActive0to2 = routeScopedCusts.filter(c => c.DaysInactive !== null && c.DaysInactive <= 2).length;
    const countWarn3to7 = routeScopedCusts.filter(c => c.DaysInactive !== null && c.DaysInactive >= 3 && c.DaysInactive <= 7).length;
    const countOver7OrNever = routeScopedCusts.filter(c => c.DaysInactive === null || c.DaysInactive > 7).length;
    const countAllOver3 = countWarn3to7 + countOver7OrNever;

    const filtered = allCusts.filter(c => { 
      const matchRoute = filterRoute === 'all' || c.RouteID === filterRoute; 
      const matchSearch = String(c.CustomerName || '').toLowerCase().includes(searchQuery) || String(c.CustomerID || '').toLowerCase().includes(searchQuery) || String(c.Phone || '').toLowerCase().includes(searchQuery); 
      if (!matchRoute || !matchSearch) return false;

      const d = c.DaysInactive;
      if (inactiveMode === 'ACTIVE_0_2') return (d !== null && d <= 2);
      if (inactiveMode === 'WARN_3_7') return (d !== null && d >= 3 && d <= 7);
      if (inactiveMode === 'CRIT_8_PLUS') return (d === null || d > 7);
      if (inactiveMode === 'OVER_3') return (d === null || d >= 3);
      return true;
    });

    // หากอยู่ในโหมดดูร้านที่ขาดส่ง ให้เรียงลำดับจากร้านที่ขาดส่งนานที่สุดขึ้นก่อน
    if (inactiveMode !== 'ALL') {
      filtered.sort((a, b) => {
        const da = a.DaysInactive === null ? 9999 : a.DaysInactive;
        const db = b.DaysInactive === null ? 9999 : b.DaysInactive;
        return db - da;
      });
    }

    const rows = filtered.map(c => { 
      const isActive = c.Status === 'Active' || !c.Status || String(c.Status).trim() === ''; 
      const statusBadge = isActive
        ? `<span class="inline-flex items-center bg-emerald-50 text-emerald-700 border border-emerald-200 px-2.5 py-1 rounded-full text-[10px] font-bold"><span class="w-1.5 h-1.5 rounded-full bg-emerald-500 mr-1.5"></span> ใช้งาน</span>`
        : `<span class="inline-flex items-center bg-rose-50 text-rose-700 border border-rose-200 px-2.5 py-1 rounded-full text-[10px] font-bold"><span class="w-1.5 h-1.5 rounded-full bg-rose-500 mr-1.5"></span> ระงับ</span>`; 

      // ⏱️ ป้ายแสดงสถานะการส่งน้ำแข็งล่าสุด & จำนวนวันที่ไม่ได้ลงน้ำแข็ง
      const days = c.DaysInactive;
      let activityBadge = '';
      if (days === null) {
        activityBadge = `
          <span class="inline-flex items-center bg-slate-100 text-slate-600 border border-slate-300 px-2.5 py-1 rounded-lg text-xs font-extrabold">⚪ ยังไม่มีประวัติซื้อ</span>
          <p class="text-[10px] text-slate-400 mt-1">ยังไม่เคยเปิดบิลในระบบ</p>
        `;
      } else if (days === 0) {
        activityBadge = `
          <span class="inline-flex items-center bg-emerald-100 text-emerald-800 border border-emerald-300 px-2.5 py-1 rounded-lg text-xs font-extrabold">🟢 ส่งแล้ววันนี้</span>
          <p class="text-[10px] text-slate-500 mt-1">ล่าสุด: ${c.LastSaleDate} (฿${Number(c.LastSaleAmount).toLocaleString()})</p>
        `;
      } else if (days <= 2) {
        activityBadge = `
          <span class="inline-flex items-center bg-emerald-50 text-emerald-700 border border-emerald-200 px-2.5 py-1 rounded-lg text-xs font-bold">🟢 ${days} วันที่แล้ว</span>
          <p class="text-[10px] text-slate-500 mt-1">ล่าสุด: ${c.LastSaleDate} (฿${Number(c.LastSaleAmount).toLocaleString()})</p>
        `;
      } else if (days >= 3 && days <= 7) {
        activityBadge = `
          <span class="inline-flex items-center bg-amber-100 text-amber-900 border-2 border-amber-400 px-2.5 py-1 rounded-lg text-xs font-black animate-pulse">⚠️ ไม่ส่ง ${days} วัน</span>

        `;
      } else {
        activityBadge = `
          <span class="inline-flex items-center bg-rose-100 text-rose-800 border-2 border-rose-400 px-2.5 py-1 rounded-lg text-xs font-black">🚨 ขาดส่ง ${days} วัน!</span>
          <p class="text-[10px] text-rose-600 font-bold mt-1">ล่าสุด: ${c.LastSaleDate} (฿${Number(c.LastSaleAmount).toLocaleString()})</p>
        `;
      }

      const promoBadge = c.HasPromo3For100 ? `<span class="inline-flex items-center bg-purple-100 border border-purple-300 text-purple-800 px-2.5 py-1 rounded-md text-[10px] font-extrabold whitespace-nowrap shadow-sm">🎁 โปรคละ 3 ชิ้น 100.-</span>` : '';

      return `
        <tr class="border-b border-slate-100 hover:bg-blue-50/30 transition text-sm">
          <td class="py-3.5 px-4 align-top">
            <div class="font-extrabold text-slate-800 text-sm cursor-pointer hover:text-blue-600 transition-colors" onclick="openCustomerModalById('${c.CustomerID}')">${c.CustomerName}</div>

          </td>
          <td class="py-3.5 px-4 align-top">
            <div class="flex items-center gap-1.5 font-bold text-indigo-700 text-xs"><i data-lucide="map-pin" class="w-3.5 h-3.5 text-indigo-400"></i> ${c.RouteName}</div>
            <div class="text-[10px] bg-slate-100 text-slate-500 border border-slate-200 inline-block px-2 py-0.5 rounded mt-1.5 font-medium">${c.CustomerType} (สะสม ${c.TotalBillsCount || 0} บิล)</div>
          </td>
          <td class="py-3.5 px-4 align-top">${activityBadge}</td>
          <td class="py-3.5 px-4 align-top min-w-[240px] max-w-[340px]">${spBadges}</td>
          <td class="py-3.5 px-4 text-center align-top pt-4">${statusBadge}</td>
          <td class="py-3.5 px-4 text-right align-top pt-3 no-print">
            <button onclick="openCustomerModalById('${c.CustomerID}')" class="p-2 bg-white border border-slate-200 text-blue-600 hover:bg-blue-50 hover:border-blue-400 rounded-lg transition shadow-sm" title="แก้ไขข้อมูล"><i data-lucide="edit-3" class="w-4 h-4"></i></button>
          </td>
        </tr>
      `;

    const routeOptions = (STATE.customerDashboard.routes || []).map(r => `<option value="${r.RouteID}" ${filterRoute === r.RouteID ? 'selected' : ''}>🚚 ${r.RouteName}</option>`).join('');

  function drawFriendlyPromotionsView() {
    const view = document.getElementById('app-view');
    const q = (document.getElementById('prm-search')?.value || '').toLowerCase();
    const filterRoute = document.getElementById('prm-route-filter')?.value || 'ALL';
    const filterMode = document.getElementById('prm-mode-filter')?.value || 'PROMO_ONLY';

    const customers = STATE.customerDashboard.list || [];
    const routes = STATE.customerDashboard.routes || [];
    const products = STATE.customerDashboard.products || [];

    // สร้างแผนที่ค้นหาลูกค้าและสินค้า (รองรับทั้ง C049 และ C-0049, P001 และ P-0001)
    const custLookup = {};
    customers.forEach(c => {
      custLookup[String(c.CustomerID).trim()] = c;
      custLookup[normCodeKey_(c.CustomerID)] = c;
    });

    const prodLookup = {};
    products.forEach(p => {
      prodLookup[String(p.ProductID).trim()] = p.ProductName;
      prodLookup[normCodeKey_(p.ProductID)] = p.ProductName;
    });

    // รวมข้อมูลจาก tb_Promotions เข้ากับรายชื่อร้านค้า
    const promoRows = (STATE.master.list || []).filter(r => r.Status !== 'Inactive');
    const promoByCustId = {};
    promoRows.forEach(r => {
      const rawName = String(r.PromoName || '');
      const rawCid = rawName.replace(/^3FOR100:/i, '').trim();
      const matchedCust = custLookup[rawCid] || custLookup[normCodeKey_(rawCid)];
      const key = matchedCust ? matchedCust.CustomerID : rawCid;
      promoByCustId[key] = {
        PromoID: r.PromoID,
        RawCid: rawCid,
        Customer: matchedCust || null,
        EligibleStr: String(r.DiscountAmt || 'ALL').trim()
      };
    });

    // สร้างรายการการ์ดแสดงผล
    let displayItems = [];
    if (filterMode === 'ALL_SHOPS') {
      displayItems = customers.map(c => {
        const prm = promoByCustId[c.CustomerID];
        return {
          PromoID: prm ? prm.PromoID : '',
          CustomerID: c.CustomerID,
          CustomerName: c.CustomerName,
          Phone: c.Phone || '',
          RouteID: c.RouteID || 'หน้าลาน',
          RouteName: c.RouteName || 'หน้าลาน',
          HasPromo: !!prm || !!c.HasPromo3For100,
          EligibleStr: prm ? prm.EligibleStr : (c.PromoEligibleProducts || 'ALL')
        };
      });
    } else {
      displayItems = Object.keys(promoByCustId).map(k => {
        const item = promoByCustId[k];
        const c = item.Customer;
        return {
          PromoID: item.PromoID,
          CustomerID: c ? c.CustomerID : item.RawCid,
          CustomerName: c ? c.CustomerName : `ร้านรหัส ${item.RawCid}`,
          Phone: c ? (c.Phone || '') : '',
          RouteID: c ? (c.RouteID || 'หน้าลาน') : '',
          RouteName: c ? (c.RouteName || 'หน้าลาน') : 'ไม่ระบุสายส่ง',
          HasPromo: true,
          EligibleStr: item.EligibleStr
        };
      });
    }

    const filtered = displayItems.filter(itm => {
      const matchRoute = filterRoute === 'ALL' || itm.RouteID === filterRoute || itm.RouteName === filterRoute;
      const matchQ = !q || String(itm.CustomerName).toLowerCase().includes(q) || String(itm.CustomerID).toLowerCase().includes(q) || String(itm.PromoID).toLowerCase().includes(q);
      return matchRoute && matchQ;

  function setReportQuickDate(mode) {
    const now = new Date();
    const fmt = d => d.toISOString().slice(0, 10);
    if (mode === 'TODAY') {
      REPORT_STATE.startDate = fmt(now);
      REPORT_STATE.endDate = fmt(now);
    } else if (mode === 'YESTERDAY') {
      const y = new Date(now);
      y.setDate(now.getDate() - 1);
      REPORT_STATE.startDate = fmt(y);
      REPORT_STATE.endDate = fmt(y);
    } else if (mode === '7DAYS') {
      const w = new Date(now);
      w.setDate(now.getDate() - 6);
      REPORT_STATE.startDate = fmt(w);
      REPORT_STATE.endDate = fmt(now);
    } else if (mode === 'MONTH') {
      const m = new Date(now.getFullYear(), now.getMonth(), 1);
      REPORT_STATE.startDate = fmt(m);
      REPORT_STATE.endDate = fmt(now);
    }
    fetchHistoricalReportData();
  }

  function onSelectReportMonth(ymStr) {
    if (!ymStr) return;
    REPORT_STATE.selectedMonthValue = ymStr;
    const parts = ymStr.split('-');
    if (parts.length === 2) {
      const yr = parseInt(parts[0], 10);
      const mo = parseInt(parts[1], 10);
      const firstDay = `${yr}-${String(mo).padStart(2, '0')}-01`;
      const lastDateObj = new Date(yr, mo, 0);
      const lastDay = `${yr}-${String(mo).padStart(2, '0')}-${String(lastDateObj.getDate()).padStart(2, '0')}`;
      REPORT_STATE.startDate = firstDay;
      REPORT_STATE.endDate = lastDay;
      REPORT_STATE.activeTab = 'CUST_RANK';
      fetchHistoricalReportData();
    }
  }

  async function fetchHistoricalReportData() {
    const sEl = document.getElementById('rpt-start');
    const eEl = document.getElementById('rpt-end');

    if (eEl) REPORT_STATE.endDate = eEl.value;

        endDate: REPORT_STATE.endDate,

      });
      REPORT_STATE.data = res;
      drawReportsCenter();

  }

  function setExpenseQuickDate(mode) {
    const now = new Date();
    const fmt = d => d.toISOString().slice(0, 10);
    if (mode === 'TODAY') {
      EXPENSE_STATE.startDate = fmt(now);
      EXPENSE_STATE.endDate = fmt(now);
    } else if (mode === 'YESTERDAY') {
      const y = new Date(now);
      y.setDate(now.getDate() - 1);
      EXPENSE_STATE.startDate = fmt(y);
      EXPENSE_STATE.endDate = fmt(y);
    } else if (mode === 'MONTH') {
      const m = new Date(now.getFullYear(), now.getMonth(), 1);
      EXPENSE_STATE.startDate = fmt(m);
      EXPENSE_STATE.endDate = fmt(now);
    }
    fetchExpenseDashboardData();
  }

  function onSelectExpenseMonth(ymStr) {
    if (!ymStr) return;
    EXPENSE_STATE.selectedMonth = ymStr;
    const parts = ymStr.split('-');
    if (parts.length === 2) {
      const yr = parseInt(parts[0], 10);
      const mo = parseInt(parts[1], 10);
      const lastDayObj = new Date(yr, mo, 0);
      EXPENSE_STATE.startDate = `${yr}-${String(mo).padStart(2, '0')}-01`;
      EXPENSE_STATE.endDate = `${yr}-${String(mo).padStart(2, '0')}-${String(lastDayObj.getDate()).padStart(2, '0')}`;
      fetchExpenseDashboardData();
    }
  }

  async function fetchExpenseDashboardData() {
    const sEl = document.getElementById('exp-start-date');
    const eEl = document.getElementById('exp-end-date');
    if (sEl) EXPENSE_STATE.startDate = sEl.value;
    if (eEl) EXPENSE_STATE.endDate = eEl.value;

    try {
      const res = await apiCall('getExpenseDashboard', {
        startDate: EXPENSE_STATE.startDate,
        endDate: EXPENSE_STATE.endDate
      });
      EXPENSE_STATE.data = res;
      drawExpenseDashboard();
    } catch (e) {}
  }

  async function cancelExpenseItem(expId) {
    if (!confirm(`⚠️ ยืนยันการยกเลิกรายการ ${expId} ใช่หรือไม่?`)) return;
    try {
      await apiCall('cancelExpense', { ExpID: expId });
      showToast('ยกเลิกรายการเรียบร้อยแล้ว');
      fetchExpenseDashboardData();
    } catch (e) {}
  }

  function appendFactoryPin(digit) {
    const inp = document.getElementById('factory-pin-input');
    if (inp && inp.value.length < 6) inp.value += digit;
  }

  function backspaceFactoryPin() {
    const inp = document.getElementById('factory-pin-input');
    if (inp) inp.value = inp.value.slice(0, -1);
  }

  function clearFactoryPin() {
    const inp = document.getElementById('factory-pin-input');
    if (inp) { inp.value = ''; inp.focus(); }
  }

  async function submitFactoryLogin() {
    const pin = document.getElementById('factory-pin-input')?.value.trim() || '';
    if (!pin) return showToast('กรุณากรอกรหัส PIN เข้าสู่ระบบ', 'error');
    try {
      const user = await apiCall('verifyFactoryLogin', { pin: pin });
      applyFactoryUserSession(user);
      sessionStorage.setItem('MAEJAI_FACTORY_USER', JSON.stringify(user));
      showToast(`ยินดีต้อนรับ ${user.EmpName} เข้าสู่ระบบค่ะ!`);
    } catch (e) {
      clearFactoryPin();
    }
  }

  function applyFactoryUserSession(user) {
    window.STATE.currentUser = user;
    document.getElementById('factory-login-screen')?.classList.add('hidden');
    if (document.getElementById('current-user-name')) {
      document.getElementById('current-user-name').innerText = user.EmpName;
    }
    if (document.getElementById('current-user-role')) {
      document.getElementById('current-user-role').innerText = `${user.Role === 'ADMIN' ? '👑 ผู้บริหาร' : '🏢 เสมียนหน้าลาน'} (${user.Position})`;
    }
    if (document.getElementById('mobile-user-badge')) {
      document.getElementById('mobile-user-badge').innerText = `👤 ${user.EmpName}`;
    }

    const empNavBtn = document.getElementById('nav-tb_Employees');
    if (empNavBtn) {
      empNavBtn.classList.toggle('hidden', user.Role !== 'ADMIN');
    }

    ensureReportsNavButton();
    lucide.createIcons();
    navigate(user.Role === 'ADMIN' ? 'dashboard' : 'pos');
  }

  function logoutFactory() {
    window.STATE.currentUser = null;
    sessionStorage.removeItem('MAEJAI_FACTORY_USER');
    if (document.getElementById('factory-pin-input')) {
      document.getElementById('factory-pin-input').value = '';
    }
    document.getElementById('factory-login-screen')?.classList.remove('hidden');
    lucide.createIcons();
  }

  window.onload = () => {
    ensureReportsNavButton();
    lucide.createIcons();
    const savedSession = sessionStorage.getItem('MAEJAI_FACTORY_USER');
    if (savedSession) {
      try {
        const user = JSON.parse(savedSession);
        if (user && user.EmpName) {
          applyFactoryUserSession(user);
          return;
        }
      } catch (e) {}
    }
  };
